#!/bin/bash

# Script to lock down file permissions, ownership, and group access on a Linux machine

LOGFILE="/var/log/lockdown_permissions.log"
BACKUP_DIR="/root/backup_permissions"
GROUPS_TO_REVIEW=("sudo" "www-data" "mysql" "docker")  # Add more groups as needed

# Function to log messages
log() {
    echo "$(date '+%Y-%m-%d %H:%M:%S') - $1" >> "$LOGFILE"
}

# Function to create backups
backup_file() {
    local file=$1
    if [[ -e $file ]]; then
        cp "$file" "$BACKUP_DIR/$(basename "$file").bak"
        log "Backed up $file to $BACKUP_DIR"
    fi
}

# Set permissions and ownership for directories
set_directory_permissions() {
    local path=$1
    echo "Setting permissions for directory $path"
    chmod 750 "$path"  # Owner can read/write/execute; group can read/execute; others none
    chown root:root "$path"  # Ensure root ownership
    log "Set permissions on $path"
}

# Set permissions and ownership for files
set_file_permissions() {
    local file=$1
    echo "Setting permissions for file $file"
    chmod 640 "$file"  # Owner can read/write; group can read; others none
    chown root:root "$file"  # Ensure root ownership
    log "Set permissions on $file"
}

# Secure sensitive directories
secure_directories() {
    log "Securing sensitive directories..."
    DIRECTORIES=(
        "/etc"
        "/root"
        "/var/log"
        "/var/lib"
        "/usr/local"
        "/srv"
        "/tmp"
    )
    
    for dir in "${DIRECTORIES[@]}"; do
        set_directory_permissions "$dir"
    done
}

# Secure common sensitive files
secure_files() {
    log "Securing sensitive files..."
    FILES=(
        "/etc/shadow"
        "/etc/gshadow"
        "/etc/passwd"
        "/etc/group"
        "/etc/ssh/sshd_config"
        "/etc/sudoers"
        "/etc/fstab"
        "/etc/hosts"
        "/etc/sysctl.conf"
        "/etc/php.ini"      # Example for PHP
        "/etc/my.cnf"      # Example for MySQL
        "/etc/nginx/nginx.conf" # Example for Nginx
    )
    
    for file in "${FILES[@]}"; do
        backup_file "$file"
        set_file_permissions "$file"
    done
    
    # Secure SSH keys for all users
    for user_home in /home/*; do
        if [ -d "$user_home/.ssh" ]; then
            chmod 700 "$user_home/.ssh"
            chmod 600 "$user_home/.ssh/id_rsa"
            chmod 644 "$user_home/.ssh/id_rsa.pub"
            chown -R "$(basename "$user_home"):$(basename "$user_home")" "$user_home/.ssh"
            log "Secured SSH keys for $user_home"
        fi
    done
}

# Lock down group memberships
lock_groups() {
    log "Locking down group memberships..."
    for group in "${GROUPS_TO_REVIEW[@]}"; do
        users_in_group=$(getent group "$group" | cut -d: -f4)
        IFS=',' read -r -a users <<< "$users_in_group"
        for user in "${users[@]}"; do
            if [[ $user != "root" ]]; then
                usermod -G "${group}" "$user"  # Keep user in the group but manage permissions
                log "Reviewed $user in group $group"
            fi
        done
    done
}

# Set ownership and permissions for specific applications
secure_application_files() {
    log "Securing application files..."
    
    # Example for web server files (adjust paths as necessary)
    WEB_DIRS=(
        "/var/www/html"
        "/var/www/logs"
    )
    
    for web_dir in "${WEB_DIRS[@]}"; do
        set_directory_permissions "$web_dir"
        chown -R www-data:www-data "$web_dir"  # Web server user and group
        log "Secured web directory $web_dir"
    done
    
    # Example for database files
    DB_DIRS=(
        "/var/lib/mysql"
        "/var/lib/postgresql"
    )
    
    for db_dir in "${DB_DIRS[@]}"; do
        set_directory_permissions "$db_dir"
        chown -R mysql:mysql "$db_dir"  # Database user and group
        log "Secured database directory $db_dir"
    done
}

# Main function to execute the lockdown
main() {
    mkdir -p "$BACKUP_DIR"
    secure_directories
    secure_files
    lock_groups
    secure_application_files

    echo "File permissions, ownership, and group access locked down successfully."
    log "Lockdown process completed."
}

# Run the main function
main
