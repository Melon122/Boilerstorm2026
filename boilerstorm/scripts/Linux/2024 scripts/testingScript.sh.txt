#!bin/bash

log() {
        local message="$1"
        local color="${2:-default}"

        case "$color" in
        red) echo -e "\033[31m$message\033[0m" ;;    # Red for errors
        green) echo -e "\033[32m$message\033[0m" ;;  # Green for success
        yellow) echo -e "\033[33m$message\033[0m" ;; # Yellow for warnings
        *) echo "$message" ;;                        # Default (no color)
        esac
}

check_root() {
        if [[ $EUID -ne 0 ]]; then
                log "Error: This script must be run as root" red
                exit 1
        fi
}

set_baseline() {
        local BASELINE_FilePath="$1"
        local BASELINE_Content="$2"
        if cp "$BASELINE_FilePath" "$BASELINE_FilePath.bak"; then
                log "Backup of $BASELINE_FilePath created at $BASELINE_FilePath.bak" green
        else
                log "Error: Failed to create backup of $BASELINE_FilePath" red
        fi

        log "$BASELINE_FilePath update start." yellow
        cat <<EOL >"$BASELINE_FilePath"
$BASELINE_Content
EOL

        if [[ $? -eq 0 ]]; then
                log "Successfully set $BASELINE_FilePath base line" green
        else
                log "Failed to set $BASELINE_FilePath base line" red
                return 1
        fi
}

fixUsers() {
        #exit on error
        set -e

        # Ensure script is run as root
        check_root

        # Define constants
        readonly FILE="correctUsers.txt"
        readonly DEFAULT_PASSWORD="Cyb3rP@triot11!"
        readonly MIN_UID=1000

        # Input validation function
        validate_username() {
                local username="$1"
                # Linux-specific username requirements
                if [[ ! "$username" =~ ^[a-z_][a-z0-9_-]*[$]?$ ]] ||
                        [[ "$username" =~ ^(root|daemon|bin|sys|system|man|lp|mail|news|uucp|proxy|www-data|backup|list|irc|gnats|nobody|_apt|admin)$ ]]; then
                        log "Invalid username format or reserved username: $username" red
                        return 1
                fi
                return 0
        }

        # Check if the file exists and is readable
        if [[ ! -f "$FILE" ]] || [[ ! -r "$FILE" ]]; then
                log "File $FILE not found or not readable!" red
                return 1
        fi

        # Secure file reading

        # Secure file reading, removing lines with "password:" and stripping "(you)"
        cleanedContent=$(sed '/password:/d; s/(you)//g' "$FILE")

        # Convert cleaned content to single-line format to ensure parsing works as expected
        singleLineContent=$(echo "$cleanedContent" | tr '\n' ' ')

        # Extract administrators and users based on the headers
        adminsLines=$(echo "$singleLineContent" | awk -F 'Authorized Administrators:' '{print $2}' | awk -F 'Authorized Users:' '{print $1}' | xargs)
        usersLines=$(echo "$singleLineContent" | awk -F 'Authorized Users:' '{print $2}' | xargs)

        # Print results for verification
        echo "Authorized Administrators: $adminsLines"

        # User confirmation
        read -p "Is this list correct? (y/n) " confirmAdmins
        if [[ "${confirmAdmins,,}" != "y" ]]; then
                log "Please correct your $FILE file" red
                exit 1
        fi

        # Makes array of adminsLines
        read -ra correctAdmins <<<"$adminsLines"

        # Extracts authorized users
        echo "Authorized Users: $usersLines"

        # User confirmation for users
        read -p "Is this list correct? (y/n) " confirmUsers
        if [[ "${confirmUsers,,}" != "y" ]]; then
                log "Please correct your $FILE file" red
                return 1
        fi

        # Makes array of usersLines
        read -ra correctUsers <<<"$usersLines"

        # Remove whitespace
        for i in "${!correctAdmins[@]}"; do
                correctAdmins[i]=$(echo "${correctAdmins[i]}" | tr -d '[:space:]')
        done
        log "Trimmed Authorized Administrators: ${correctAdmins[@]}" yellow

        for i in "${!correctUsers[@]}"; do
                correctUsers[i]=$(echo "${correctUsers[i]}" | tr -d '[:space:]')
        done
        log "Trimmed Authorized Users: ${correctUsers[@]}" yellow

        # Validate usernames
        for admin in "${correctAdmins[@]}"; do
                if ! validate_username "$admin"; then
                        log "Invalid administrator username detected. Aborting." red
                        return 1
                fi
        done

        for user in "${correctUsers[@]}"; do
                if ! validate_username "$user"; then
                        log "Invalid user username detected. Aborting." red
                        return 1
                fi
        done

        # Get all users from the passwd file
        currentUsers=($(awk -F: '$3 >= 1000 && $1 != "nobody" {print $1}' /etc/passwd))

        # Add users that are not already on the system.
        for user in "${correctUsers[@]}"; do
                if [[ ! " ${currentUsers[*]} " =~ " ${user} " ]]; then
                        if ! useradd \
                                --create-home \
                                --shell /bin/bash \
                                --comment "Managed user" \
                                --password "$(openssl passwd -6 "$DEFAULT_PASSWORD")" \
                                "$user"; then
                                log "Failed to add user: $user" red
                                continue
                        fi
                        log "Added user: $user" yellow
                fi
        done

        # Get an array of all admins
        currentAdmins=($(grep '^sudo:.*$' /etc/group | awk -F: '{print $4}' | tr ',' ' '))

        # Compare and add new admins or convert existing users to admins
        for admin in "${correctAdmins[@]}"; do
                if [[ ! " ${currentAdmins[*]} " =~ " ${admin} " ]]; then
                        if [[ " ${currentUsers[*]} " =~ " ${admin} " ]]; then
                                usermod -aG sudo "$admin"
                                log "Converted user to admin: $admin" yellow
                        else
                                if ! useradd \
                                        --create-home \
                                        --shell /bin/bash \
                                        --comment "Managed administrator" \
                                        --password "$(openssl passwd -6 "$DEFAULT_PASSWORD")" \
                                        "$admin"; then
                                        echo "Failed to add admin: $admin" >&2
                                        continue
                                fi
                                usermod -aG sudo "$admin"
                                log "Added admin: $admin" yellow
                        fi
                fi
        done

        # Compare and remove users
        for user in "${currentUsers[@]}"; do
                if [[ ! " ${correctUsers[*]} " =~ " ${user} " ]] && [[ ! " ${correctAdmins[*]} " =~ " ${user} " ]]; then
                        if ! deluser --remove-home "$user" 2>/dev/null; then
                                log "Failed to remove user: $user" red
                                continue
                        fi
                        log "Deleted user and their files: $user" yellow
                fi
        done

        # Compare and remove admins or convert admins to users
        for admin in "${currentAdmins[@]}"; do
                if [[ ! " ${correctAdmins[*]} " =~ " ${admin} " ]]; then
                        if [[ " ${correctUsers[*]} " =~ " ${admin} " ]]; then
                                deluser "$admin" sudo
                                log "Converted admin to user: $admin" yellow
                        else
                                if ! deluser --remove-home "$admin" 2>/dev/null; then
                                        log "Failed to remove admin: $admin" red
                                        continue
                                fi
                                log "Removed admin: $admin" yellow
                        fi
                fi
        done

        # Set secure password policies
        for admin in "${correctAdmins[@]}"; do
                # Set password aging policies
                chage -m 15 -M 60 -I 5 -W 7 "$admin"

                # Set strong password with salt (hashed)
                hashed_password=$(openssl passwd -6 "$DEFAULT_PASSWORD")
                usermod --password "$hashed_password" "$admin" || {
                        log "Failed to set password for admin: $admin" red
                }
        done

        for user in "${correctUsers[@]}"; do
                # Set password aging policies
                chage -m 15 -M 60 -I 5 -W 7 "$user"

                # Set strong password with salt (hashed)
                hashed_password=$(openssl passwd -6 "$DEFAULT_PASSWORD")
                usermod --password "$hashed_password" "$user" || {
                        log "Failed to set password for user: $user" red
                }
        done
}

        validate_param() {
                local param="$1"
                if ! sysctl -q "$param" &>/dev/null; then
                        log "Warning: Parameter $param is not valid on this system" yellow
                        return 1
                fi
                return 0

        # Function to check if parameter already exists in config
        check_existing() {
                local param="$1"
                local config_file="$2"
                if grep -q "^${param}\s*=" "$config_file" 2>/dev/null; then
                        return 0
                fi
                return 1

                        update_config "$config" "$config_file"
                done

                read -p "Do you want to configure IPv6? (y/n): " -r ipv6_needed
                ipv6_needed="${ipv6_needed,,}"

                if [[ "$ipv6_needed" == "y" ]]; then
                        ipv6_configurations=(
                                'net.ipv6.conf.all.accept_ra=0'
                                'net.ipv6.conf.all.accept_ra_pinfo=0'
                                'net.ipv6.conf.all.accept_redirects=0'
                                'net.ipv6.conf.all.accept_source_route=0'
                                'net.ipv6.conf.all.autoconf=0'
                                'net.ipv6.conf.all.dad_transmits=0'
                                'net.ipv6.conf.all.forwarding=0'
                                'net.ipv6.conf.all.max_addresses=1'
                                'net.ipv6.conf.all.use_tempaddr=2'
                                'net.ipv6.conf.default.accept_ra=0'
                                'net.ipv6.conf.default.accept_ra_pinfo=0'
                                'net.ipv6.conf.default.accept_redirects=0'
                                'net.ipv6.conf.default.accept_source_route=0'
                                'net.ipv6.conf.default.autoconf=0'
                                'net.ipv6.conf.default.dad_transmits=0'
                                'net.ipv6.conf.default.forwarding=0'
                                'net.ipv6.conf.default.max_addresses=1'
                                'net.ipv6.conf.default.use_tempaddr=2'         # Disable temporary addresses
                                'net.ipv6.conf.all.disable_ipv6=1'             # Disable IPv6 on all interfaces
                                'net.ipv6.conf.default.disable_ipv6=1'         # Disable IPv6 on the default interface
                                'net.ipv6.conf.lo.disable_ipv6=1'              # Disable IPv6 on the loopback interface
                                'net.ipv6.conf.default.router_solicitations=0' # Disable router solicitations
                                'net.ipv6.conf.default.accept_ra_rtr_pref=0'   # Disable accepting router preference
                                'net.ipv6.conf.default.accept_ra_defrtr=0'     # Disable accepting default routers
                                'net.ipv6.conf.all.rp_filter=1'                # Enable reverse path filtering

                        )
                        log "Configuring IPv6 settings" green
                        for config in "${ipv6_configurations[@]}"; do
                                log "Adding IPv6 configuration: $config" green
                                update_config "$config" "$config_file"
                        done
                else
                        log "Disabled IPv6 configurations" green
                        ipv6_configurations=(
                                'net.ipv6.conf.default.router_solicitations=0'
                                'net.ipv6.conf.default.accept_ra_rtr_pref=0'
                                'net.ipv6.conf.default.accept_ra_pinfo=0'
                                'net.ipv6.conf.default.accept_ra_defrtr=0'
                                'net.ipv6.conf.all.accept_ra=0'
                                'net.ipv6.conf.default.accept_ra=0'
                                'net.ipv6.conf.default.autoconf=0'
                                'net.ipv6.conf.default.dad_transmits=0'
                                'net.ipv6.conf.default.max_addresses=1'
                                'net.ipv6.conf.all.rp_filter=1'
                                'net.ipv6.conf.all.forwarding=0'
                        )
                        for config in "${ipv6_configurations[@]}"; do
                                # Disable IPv6 by modifying parameters
                                update_config "$config" "$config_file"
                        done
                fi

                # Apply changes after writing to the file
                if ! sysctl -p "$config_file"; then
                        log "Error: Failed to apply some sysctl parameters" red
                fi

                log "SysCtl Hardening completed"
        }

        # Execute main function
        main "$@"
}

setup_ufw() {
        # Update package lists
        if apt update; then
                log "Package lists updated successfully." green
        else
                log "Failed to update package lists." red
                return 1
        fi

        # Install and or Upgrade UFW
        if apt install -y --only-upgrade ufw; then
                log "UFW installed successfully." green
        else
                log "Failed to install UFW." red
                return 1
        fi

        # Set default policies
        sudo ufw default deny incoming
        log "Set default policy to deny incoming connections." green

        sudo ufw default allow outgoing
        log "Set default policy to allow outgoing connections." green

        # Enable UFW
        if sudo ufw enable; then
                log "UFW enabled successfully." green
        else
                log "Failed to enable UFW." red
                return 1
        fi

        # Enable UFW to start on boot
        if sudo systemctl enable ufw; then
                log "UFW will start on boot." green
        else
                log "Failed to enable UFW to start on boot." red
                return 1
        fi

        # Show UFW status
        sudo ufw status verbose

        # Enable logging
        if sudo ufw logging on; then
                log "UFW logging enabled." green
        else
                log "Failed to enable UFW logging." red
                return 1
        fi

        # Limit SSH connections
        if sudo ufw limit ssh; then
                log "SSH connections are now limited." green
        else
                log "Failed to limit SSH connections." red
                return 1
        fi
}

pam_config() {

        # Install required PAM modules
        echo "Installing necessary PAM modules..."
        sudo apt-get update
        sudo apt-get install -y libpam-pwquality libpam-fail2ban libpam-unix libpam-systemd libpam-modules

        # Ensure cracklib is removed to avoid conflicts with pwquality
        echo "Removing libpam-cracklib to avoid conflicts..."
        sudo apt-get remove -y libpam-cracklib

        declare -A pam_files
        pam_files=(
                ["/etc/pam.d/common-auth"]="
auth required pam_faildelay.so delay=5000000
auth required pam_faillock.so preauth
auth [success=1 default=ignore] pam_unix.so
auth [default=die] pam_faillock.so authfail
auth sufficient pam_faillock.so authsucc
auth requisite pam_deny.so
auth required pam_permit.so
        "

                ["/etc/pam.d/common-account"]="
account [success=1 new_authtok_reqd=done default=ignore]        pam_unix.so
account requisite               pam_deny.so
account required                pam_access.so
account required                pam_permit.so
        "

                ["/etc/pam.d/common-password"]="
password        required                        pam_pwquality.so difok=8 minlen=14 dcredit=-1 ucredit=-1 lcredit=-1 ocredit=-1 minclass=4 maxrepeat=2 gecoscheck=1 dictcheck=1 maxsequence=4 maxclassrepeat=4 usercheck=1 enforcing=1 enforce_for_root=1 usercheck=1 usersubstr=4 retry=2
password        requisite                       pam_pwhistory.so remember=5 use_authtok
password        [success=1 default=ignore]  pam_unix.so yescrypt try_first_pass obscure shadow rounds=11
password        requisite                       pam_deny.so
password        required                        pam_permit.so
        "

                ["/etc/pam.d/common-session"]="
session         required                                        pam_limits.so
session [default=1]             pam_permit.so
session requisite               pam_deny.so
session required                pam_permit.so
session optional                pam_umask.so
session required        pam_unix.so
session optional        pam_systemd.so
        "

                ["/etc/pam.d/common-session-noninteractive"]="
session [default=1]             pam_permit.so
session requisite               pam_deny.so
session required                pam_permit.so
session optional                pam_umask.so
session required                pam_unix.so
        "
        )

        # Iterate over the array and call the set_pam_file function
        for file in "${!pam_files[@]}"; do
                set_baseline "$file" "${pam_files[$file]}"
                if [[ $? -ne 0 ]]; then
                        log "An error occurred while updating $file" red
                fi
        done
}

auto_upgrade() {

        log "Configuring unattended upgrades for automatic updates..." green

        # Install unattended-upgrades if not already installed
        if ! apt list --installed | grep -q unattended-upgrades; then
                log "Installing unattended-upgrades package..." green
                sudo apt-get update
                sudo apt-get install -y unattended-upgrades
        fi

        set_baseline "/etc/apt/apt.conf.d/20auto-upgrades" "
# Enable automatic upgrades
APT::Periodic::Update-Package-Lists "1";
APT::Periodic::Unattended-Upgrade "1";"

        set_baseline "/etc/apt/apt.conf.d/50unattended-upgrades" "APT::Periodic::Update-Package-Lists "1";

APT::Periodic::Unattended-Upgrade "1";
APT::Periodic::Verbose "3";
APT::Periodic::RandomSleep "1800";
APT::Periodic::Unattended-Upgrade::Automatic-Reboot "0";
APT::Sandbox::Seccomp "true";
APT::Sandbox::ResetEnvironment "true";
APT::Get::AllowUnauthenticated "false";
Acquire::ForceHash "sha256";
Acquire::AllowInsecureRepositories "0";
Acquire::AllowDowngradeToInsecureRepositories "0";
Acquire::AllowWeakRepositories "0";
Unattended-Upgrade::Remove-Unused-Dependencies "true";
Unattended-Upgrade::Remove-Unused-Kernel-Packages "true";
Unattended-Upgrade::Allowed-Origins {
        "${distro_id} stable";

        "${distro_id} ${distro_codename}-updates";
};
Unattended-Upgrade::Automatic-Reboot "false";
APT::Periodic::AutocleanInterval "1";"

        # Restart the unattended-upgrades service

}

# Main function to delete files
deleteFiles() {

        # Function to confirm file deletion
        confirm_delete() {
                read -p "Do you want to delete '$1'? (y/n): " file_confirmation
                if [[ "$file_confirmation" =~ [Yy] ]]; then
                        rm -f "$1"
                        log "File '$1' deleted." yellow
                else
                        log "File '$1' not deleted." yellow
                fi
        }

        log "Starting file deletion process..." green

        # Find and prompt for deleting files with no owner
        log "Searching for files with no owner..." yellow
        local no_owner_files
        no_owner_files=$(df --local -P | awk '{if (NR!=1) print $6}' | xargs -I '{}' find '{}' -xdev -nouser)

        if [ -n "$no_owner_files" ]; then
                echo "$no_owner_files" | while read -r file; do
                        confirm_delete "$file"
                done
        else
                log "No files with no owner found." yellow
        fi

        # Find specific file types
        log "Searching for specific document files (*.xlsx, *.docx)..." yellow
        local document_files
        document_files=$(sudo find / -type f \( -name "*.xlsx" -o -name "*.docx" -o -name "*.csv" \) 2>/dev/null)

        if [ -n "$document_files" ]; then
                echo "$document_files" | while read -r file; do
                        log "Found document file: $file" yellow
                done
        else
                log "No document files found." yellow
        fi

        # Find suspicious .txt files
        log "Finding suspicious .txt files..." yellow
        local suspicious_txt_files
        suspicious_txt_files=$(
                sudo find / -type f \( -name ".*.txt" -o -name "*.txt" \) -not \( -path "/usr/lib/python3*" -o -path "/snap/core20*" -o -path "/usr/lib/firmware*"
                -o -path "/usr/share/gnupg*" -o -path "/usr/share/snmp*" -o -path "/snap/core22*" -o -path "/usr/share/doc*" -o -path "/snap/snap-store*" -o -path "/snap/gnome*" \) 2>/
                d
                ev/null
        )
        if [ -n "$suspicious_txt_files" ]; then
                echo "$suspicious_txt_files" | while read -r file; do
                        confirm_delete "$file"
                done
        else
                log "No suspicious .txt files found." yellow
        fi

        # Find and delete media files with confirmation
        log "Searching for media files..." yellow
        find . \( -name "*.gif" -o -name "*.png" -o -name "*.jpg" -o -name "*.jpeg" -o -name "*.bmp" -o -name "*.mp4" -o -name "*.mov" -o -name "*.wmv" -o -name "*.avi" -o -name "*.mkv" -o -name "*.mp3" -o -name "*.aac" -o -name "*.flac" -o -name "*.alac" -o -name "*.wav" \) \
                -not -path "/usr/share/icons/*" \
                -not -path "/usr/share/pixmaps/*" \
                -not -path "/usr/share/backgrounds/*" \
                -not -path "/usr/share/doc/*" \
                -not -path "/usr/share/locale/*" \
                -not -path "/usr/share/fonts/*" \
                -not -path "/usr/share/themes/*" \
                -not -path "/usr/share/sounds/*" \
                -print0 | while IFS= read -r -d $'\0' file; do
                confirm_delete "$file"
        done

        log "Searching for more suspicious files..." yellow
        # Search for files containing specified keywords
        local keywords=("shell" "secret" "password" "pw" "games" "users" "ploit" "sniff" "crack" "malware" "rat" "backdoor" "rootkit" "inject" "hack" "keylogger" "trojan" "attac
k" "hash" "vuln" "ware" "brute" "payload" "dump" "phish")
        for keyword in "${keywords[@]}"; do
                log "Searching for files containing the keyword '$keyword'..." yellow
                find . -type f -exec grep -li "$keyword" {} + 2>/dev/null | while read -r file; do
                        log "Suspicious file found: $file" red
                        confirm_delete "$file"
                done
        done

        log "File deletion process completed." green
}

secure_permissions_ownership() {
        LOGFILE="/var/log/lockdown_permissions.log"
        BACKUP_DIR="/root/backup_permissions"
        GROUPS_TO_REVIEW=("sudo" "adm" "cdrom" "plugdev" "www-data" "ssh" "systemd-journal") # Common groups in a baseline Ubuntu installation

        mkdir -p "$(dirname "$LOGFILE")"

        # Function to create backups
        backup_file() {
                local file=$1
                if [[ -e $file ]]; then
                        cp "$file" "$BACKUP_DIR/$(basename "$file").bak" && log "Backed up $file to $BACKUP_DIR" green || log "Error backing up $file" red
                else
                        log "File $file does not exist, skipping backup." yellow
                fi
        }

        # Set permissions and ownership for directories
        set_directory_permissions() {
                local path=$1
                echo "Setting permissions for directory $path"
                chmod 750 "$path" && log "Set permissions on $path" green || log "Failed to set permissions on $path" red
                chown root:root "$path" && log "Set ownership on $path" green || log "Failed to set ownership on $path" red
        }

        # Set permissions and ownership for files
        set_file_permissions() {
                local file=$1
                echo "Setting permissions for file $file"
                chmod 640 "$file" && log "Set permissions on $file" green || log "Failed to set permissions on $file" red
                chown root:root "$file" && log "Set ownership on $file" green || log "Failed to set ownership on $file" red
        }

        # Secure sensitive directories
        secure_directories() {
                log "Securing sensitive directories..."
                DIRECTORIES=(
                        "/etc"
                        "/root"
                        "/var/log"
                        "/var/lib"
                        "/usr/local"
                        "/srv"
                        "/tmp"
                )

                for dir in "${DIRECTORIES[@]}"; do
                        if [[ -d $dir ]]; then
                                set_directory_permissions "$dir"
                        else
                                log "Directory $dir does not exist, skipping." yellow
                        fi
                done
        }

        # Secure common sensitive files
        secure_files() {
                log "Securing sensitive files..."
                FILES=(
                        "/etc/shadow"                     # User passwords and account information
                        "/etc/gshadow"                    # Group passwords
                        "/etc/passwd"                     # User account information
                        "/etc/group"                      # Group account information
                        "/etc/ssh/sshd_config"            # SSH server configuration
                        "/etc/sudoers"                    # Sudo permissions
                        "/etc/fstab"                      # File system mount points
                        "/etc/hosts"                      # Hostname resolution
                        "/etc/sysctl.conf"                # Kernel parameters at boot
                        "/etc/php.ini"                    # PHP configuration file
                        "/etc/my.cnf"                     # MySQL configuration file
                        "/etc/nginx/nginx.conf"           # Nginx configuration file
                        "/etc/apache2/apache2.conf"       # Apache configuration file
                        "/etc/postgresql/postgresql.conf" # PostgreSQL configuration file
                        "/etc/mysql/my.cnf"               # MySQL configuration
                        "/etc/default/ufw"                # Uncomplicated Firewall settings
                        "/etc/network/interfaces"         # Network interface configurations
                        "/etc/resolv.conf"                # DNS resolver configuration
                        "/etc/hostname"                   # System hostname
                        "/etc/hosts.allow"                # TCP Wrappers access control
                        "/etc/hosts.deny"                 # TCP Wrappers deny rules
                        "/etc/crontab"                    # System-wide cron jobs
                        "/etc/security/access.conf"       # Access control rules
                        "/etc/pam.d/common-auth"          # PAM authentication configuration
                        "/etc/pam.d/common-password"      # PAM password policies
                        "/etc/pam.d/common-session"       # PAM session management
                        "/etc/shells"                     # Allowed login shells
                        "/etc/cron.allow"                 # Allowed cron users
                        "/etc/cron.deny"                  # Denied cron users
                        "/var/log/auth.log"               # Authentication logs
                        "/var/log/syslog"                 # System logs
                        "/var/log/secure"                 # Security logs (if applicable)
                )

                for file in "${FILES[@]}"; do
                        backup_file "$file"
                        set_file_permissions "$file"
                done

                # Secure SSH keys for all users
                for user_home in /home/*; do
                        if [ -d "$user_home/.ssh" ]; then
                                chmod 700 "$user_home/.ssh" && log "Secured .ssh directory for $user_home" green || log "Failed to secure .ssh directory for $user_home" red
                                chmod 600 "$user_home/.ssh/id_rsa" && log "Secured id_rsa for $user_home" green || log "Failed to secure id_rsa for $user_home" red
                                chmod 644 "$user_home/.ssh/id_rsa.pub" && log "Secured id_rsa.pub for $user_home" green || log "Failed to secure id_rsa.pub for $user_home" red
                                chown -R "$(basename "$user_home"):$(basename "$user_home")" "$user_home/.ssh" && log "Set ownership for .ssh directory of $user_home" green || log "Failed to set own
ership for .ssh directory of $user_home" red
                        fi
                done
        }

        # Lock down group memberships
        lock_groups() {
                log "Locking down group memberships..."
                for group in "${GROUPS_TO_REVIEW[@]}"; do
                        users_in_group=$(getent group "$group" | cut -d: -f4)
                        IFS=',' read -r -a users <<<"$users_in_group"
                        for user in "${users[@]}"; do
                                if [[ $user != "root" ]]; then
                                        usermod -G "${group}" "$user" && log "Reviewed $user in group $group" green || log "Failed to modify group for $user" red
                                fi
                        done
                done
        }

        # Set ownership and permissions for specific applications
        secure_application_files() {
                log "Securing application files..."

                WEB_DIRS=(
                        "/var/www/html"
                        "/var/www/logs"
                )

                for web_dir in "${WEB_DIRS[@]}"; do
                        if [[ -d $web_dir ]]; then
                                set_directory_permissions "$web_dir"
                                chown -R www-data:www-data "$web_dir" && log "Secured web directory $web_dir" green || log "Failed to secure web directory $web_dir" red
                        else
                                log "Web directory $web_dir does not exist, skipping." yellow
                        fi
                done

                DB_DIRS=(
                        "/var/lib/mysql"
                        "/var/lib/postgresql"
                )

                for db_dir in "${DB_DIRS[@]}"; do
                        if [[ -d $db_dir ]]; then
                                set_directory_permissions "$db_dir"
                                chown -R mysql:mysql "$db_dir" && log "Secured database directory $db_dir" green || log "Failed to secure database directory $db_dir" red
                        else
                                log "Database directory $db_dir does not exist, skipping." yellow
                        fi
                done
        }

        # Main function to execute the lockdown
        main() {
                mkdir -p "$BACKUP_DIR" && log "Backup directory created at $BACKUP_DIR" green || log "Failed to create backup directory at $BACKUP_DIR" red
                secure_directories
                secure_files
                lock_groups
                secure_application_files

                log "File permissions, ownership, and group access locked down successfully." green
        }
        main "$@"
}

# Function to check cron jobs
check_cron_jobs() {
        log "Checking system cron jobs for integrity and security..." yellow

        local cron_files=(
                "/etc/crontab"               # System-wide crontab
                "/etc/cron.d/*"              # Cron jobs defined in the cron.d directory
                "/var/spool/cron/crontabs/*" # User-specific crontab files
                "/etc/anacrontab"            # Anacron configuration file
                "/etc/cron.hourly/*"         # Scripts to run hourly
                "/etc/cron.daily/*"          # Scripts to run daily
                "/etc/cron.weekly/*"         # Scripts to run weekly
                "/etc/cron.monthly/*"        # Scripts to run monthly
        )

        for cron_file in "${cron_files[@]}"; do
                if [ -e $cron_file ]; then
                        log "Inspecting $cron_file..." green

                        if grep -E 'shell|bash|sh|perl|python|php|wget|curl|nc|netcat|scp|ftp|ruby|node|java|git|tcl|lsof|curl|openssl|rsync|telnet|ssh|chmod|chown|dd' "$cron_file"; then
                                log "Warning: Potentially suspicious commands found in $cron_file" yellow
                        fi

                        if [ "$(stat -c %A $cron_file | cut -c 5)" == "w" ]; then
                                log "Warning: $cron_file is world-writable!" red
                        fi

                        if ! crontab -l -u "$(basename $cron_file)" >/dev/null 2>&1; then
                                log "Error: Syntax error detected in $cron_file" red
                        fi

                else
                        log "$cron_file does not exist." yellow
                fi
        done

        if systemctl is-active cron >/dev/null 2>&1; then
                log "Cron service is running." green
        else
                log "Cron service is not running!" red
                sudo systemctl start cron
                sudo systemctl enable cron
        fi
}

# Function to check network connections
check_network_connections() {
        log "Checking for suspicious network connections..." yellow

        # Capture all network connections, including listening and established states
        local network_connections=$(netstat -tulnp 2>/dev/null)

        # Check for listening connections on all interfaces (0.0.0.0 and ::)
        local suspicious_listening=$(echo "$network_connections" | grep -E '0.0.0.0|::' | grep LISTEN)

        if [[ -n $suspicious_listening ]]; then
                log "Suspicious listening connections detected:\n$suspicious_listening" red
        else
                log "No suspicious listening connections found." green
        fi

        # Check for established connections to external IPs
        local suspicious_established=$(echo "$network_connections" | grep ESTABLISHED | grep -v '127.0.0.1')

        if [[ -n $suspicious_established ]]; then
                log "Suspicious established connections detected:\n$suspicious_established" red
        else
                log "No suspicious established connections found." green
        fi

        # Check for connections on unusual or high-risk ports
        local high_risk_ports=(23 25 80 110 143 443 3306 5432 6666 31337)
        for port in "${high_risk_ports[@]}"; do
                local connections_on_port=$(echo "$network_connections" | grep ":$port ")
                if [[ -n $connections_on_port ]]; then
                        log "Connections found on high-risk port $port:\n$connections_on_port" red
                fi
        done

        # Summary of network connections checked
        log "Network connection check completed." green
}

troubleshoot_network() {
        # Function to check network interfaces
        check_network_interfaces() {
                log "Checking network interfaces..." yellow
                ip a | grep -E "^[0-9]+: " || log "No network interfaces found." red
        }

        # Function to check DNS resolution
        check_dns() {
                log "Checking DNS resolution..." yellow
                local test_domain="google.com"
                if nslookup "$test_domain" >/dev/null 2>&1; then
                        log "DNS resolution for $test_domain is successful." green
                else
                        log "DNS resolution for $test_domain failed." red
                fi
        }

        # Function to check connectivity using ping
        check_ping() {
                log "Pinging external address..." yellow
                local test_ip="8.8.8.8" # Google's public DNS
                if ping -c 4 "$test_ip" >/dev/null 2>&1; then
                        log "Ping to $test_ip is successful." green
                else
                        log "Ping to $test_ip failed." red
                fi
        }

        # Function to check current network connections
        check_connections() {
                log "Checking current network connections..." yellow
                ss -tulnp || log "Failed to retrieve network connections." red
        }

        # Function to check for firewall rules
        check_firewall() {
                log "Checking firewall status..." yellow
                if command -v ufw >/dev/null; then
                        ufw status verbose || log "Failed to retrieve firewall status." red
                else
                        log "UFW is not installed. Please check the firewall settings manually." yellow
                fi
        }

        # Function to display routing table
        check_routing_table() {
                log "Checking routing table..." yellow
                route -n || log "Failed to retrieve routing table." red
        }

        # Main function to run all checks
        main() {
                log "Starting network troubleshooting..." green
                check_network_interfaces
                check_dns
                check_ping
                check_connections
                check_firewall
                check_routing_table
                log "Network troubleshooting completed." green
        }

        main
}

        # Define suspicious and unusual ports
        local suspicious_ports=(
                22   # SSH

                8080 # Alternative HTTP
                8888 # Web application
                53   # DNS
                25   # SMTP
                110  # POP3
                143  # IMAP
        )

        # Define unusual ports for additional scrutiny
        local unusual_ports=(
                6666  # Often used for various services
                31337 # Back Orifice (potentially malicious)
                9999  # Commonly used for various applications
                4444  # Often used for backdoor access
                5200  # May indicate a rogue service
                5000  # Commonly used by UPnP
        )

        # Check for sockets on suspicious and unusual ports
        for port in "${suspicious_ports[@]}" "${unusual_ports[@]}"; do
                echo "$socket_info" | grep -E "$port" | while read -r line; do
                        log "Suspicious or unusual socket on port $port: $line" red
                done
        done

        # Check for sockets with unusual states
        local unusual_states=("TIME_WAIT" "CLOSE_WAIT" "SYN_RECV")
        for state in "${unusual_states[@]}"; do
                echo "$socket_info" | grep "$state" | while read -r line; do
                        log "Unusual socket state detected: $line" red
                done
        done

        # Check for listening sockets without associated processes
        echo "$socket_info" | grep LISTEN | while read -r line; do
                if [[ $line != *"LISTEN"* ]]; then
                        log "Listening socket without an associated process: $line" red
                fi
        done

        log "Socket check completed." green
}

# Function to check running processes
check_processes() {
        log "Checking for suspicious running processes..." yellow

        # Define an array of suspicious keywords
        local suspicious_keywords=(
                'bash' 'sh' 'perl' 'python' 'php' 'wget' 'curl' 'nc'
                'netcat' 'scp' 'ftp' 'tclsh' 'ruby' 'powershell' 'java'
                'exec' 'shell' 'sh' 'node' 'socket' 'ssh' 'httpd' 'nginx'
                'mysqld' 'postgres' 'java' 'mongod' 'redis-server' 'sql' 'backdoor'
                'sploit'
        )

        # Construct the grep pattern from the keywords
        local grep_pattern=$(
                IFS=\|
                echo "${suspicious_keywords[*]}"
        )

        # Check for suspicious processes
        ps aux | grep -E "$grep_pattern" | while read -r process; do
                log "Suspicious process found: $process" red
        done

        # Additional checks for common indicators of compromise
        log "Checking for processes running from unusual locations..." yellow
        ps aux | awk '$11 !~ /^(/ {print $0}' | while read -r process; do
                log "Unusual process path detected: $process" red
        done

        # Check for processes with high CPU or memory usage
        log "Checking for processes consuming excessive resources..." yellow
        ps aux --sort=-%cpu | awk 'NR<=5 {print}' | while read -r process; do
                log "High CPU usage process: $process" red
        done
        ps aux --sort=-%mem | awk 'NR<=5 {print}' | while read -r process; do
                log "High memory usage process: $process" red
        done

        log "Process check completed." green
}

# Function to validate input
validate_input() {

                        fi
                done
                return 0
        fi
        return 1
}
